  document.querySelectorAll("[data-api-code-preview]").forEach(function (root) {
    var scenarios = [];
    try {
      scenarios = JSON.parse(root.getAttribute("data-scenarios") || "[]");
    } catch {
      return;
    }
    if (!scenarios.length) return;

    var endpointEl = root.querySelector("[data-api-endpoint]");
    var jsonEl = root.querySelector("[data-api-json]");
    if (!endpointEl || !jsonEl) return;

    var reduced =
      window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    var current = 0;
    var stopped = false;
    var loopPromise = null;
    var reducedInterval = null;

    var ERASE_MS = 28;
    var TYPE_MS = 38;
    var HOLD_MS = 3200;

    function sleep(ms) {
      return new Promise(function (resolve) {
        window.setTimeout(resolve, ms);
      });
    }

    function setScenario(index) {
      var scenario = scenarios[index];
      endpointEl.textContent = scenario.endpoint;
      jsonEl.textContent = scenario.json;
      current = index;
    }

    function eraseText(text) {
      return new Promise(function (resolve) {
        var value = text;
        function step() {
          if (stopped || value.length === 0) {
            endpointEl.textContent = value;
            resolve(value);
            return;
          }
          value = value.slice(0, -1);
          endpointEl.textContent = value;
          window.setTimeout(step, ERASE_MS);
        }
        step();
      });
    }

    function typeText(text) {
      return new Promise(function (resolve) {
        var index = 0;
        function step() {
          if (stopped) {
            resolve();
            return;
          }
          index += 1;
          endpointEl.textContent = text.slice(0, index);
          if (index >= text.length) {
            resolve();
            return;
          }
          window.setTimeout(step, TYPE_MS);
        }
        step();
      });
    }

    async function runCycle() {
      while (!stopped) {
        await sleep(HOLD_MS);
        if (stopped) break;

        var next = (current + 1) % scenarios.length;
        await eraseText(scenarios[current].endpoint);
        if (stopped) break;

        jsonEl.textContent = scenarios[next].json;
        await typeText(scenarios[next].endpoint);
        if (stopped) break;

        current = next;
      }
    }

    function startReduced() {
      setScenario(0);
      if (reducedInterval) return;
      reducedInterval = window.setInterval(function () {
        setScenario((current + 1) % scenarios.length);
      }, HOLD_MS);
    }

    function stopReduced() {
      if (!reducedInterval) return;
      window.clearInterval(reducedInterval);
      reducedInterval = null;
    }

    function start() {
      if (reduced) {
        startReduced();
        return;
      }
      if (loopPromise) return;
      stopped = false;
      loopPromise = runCycle().finally(function () {
        loopPromise = null;
      });
    }

    document.addEventListener("visibilitychange", function () {
      if (document.hidden) {
        stopped = true;
        stopReduced();
      } else {
        stopped = false;
        start();
      }
    });

    start();
  });
